   1: import * as THREE from "three";
   2: import { CLIP_DURATIONS } from "./SceneIntegrator";
   3: 
   4: export interface QueueSegment {
   5:   clip: string;
   6:   from: number;
   7:   to: number;
   8:   elapsed: number;
   9: }
  10: 
  11: export type DirectorMode = "coding" | "sequence" | "safe-return" | "settled";
  12: 
  13: export class CharacterDirector {
  14:   private avatarMixer: THREE.AnimationMixer;
  15:   private chairMixer: THREE.AnimationMixer;
  16:   private avatarActions: Record<string, THREE.AnimationAction>;
  17:   private chairActions: Record<string, THREE.AnimationAction>;
  18:   private bodyTurn: THREE.Object3D;
  19:   private chairRoot: THREE.Object3D;
  20:   private restBody: THREE.Quaternion;
  21:   private restChair: THREE.Quaternion;
  22: 
  23:   public mode: DirectorMode = "coding";
  24:   public currentClip: string = "coding_idle";
  25:   public currentTime: number = 0;
  26:   private queue: QueueSegment[] = [];
  27:   public revision: number = 0;
  28: 
  29:   constructor(
  30:     avatarMixer: THREE.AnimationMixer,
  31:     chairMixer: THREE.AnimationMixer,
  32:     avatarActions: Record<string, THREE.AnimationAction>,
  33:     chairActions: Record<string, THREE.AnimationAction>,
  34:     bodyTurn: THREE.Object3D,
  35:     chairRoot: THREE.Object3D
  36:   ) {
  37:     this.avatarMixer = avatarMixer;
  38:     this.chairMixer = chairMixer;
  39:     this.avatarActions = avatarActions;
  40:     this.chairActions = chairActions;
  41:     this.bodyTurn = bodyTurn;
  42:     this.chairRoot = chairRoot;
  43: 
  44:     this.restBody = bodyTurn.getWorldQuaternion(new THREE.Quaternion());
  45:     this.restChair = chairRoot.getWorldQuaternion(new THREE.Quaternion());
  46: 
  47:     this.applyClip("coding_idle", 0);
  48:   }
  49: 
  50:   private applyClip(clip: string, time: number) {
  51:     const duration = CLIP_DURATIONS[clip];
  52:     if (duration === undefined) {
  53:       throw new Error(`Unknown animation clip: ${clip}`);
  54:     }
  55:     const t = THREE.MathUtils.clamp(time, 0, duration);
  56: 
  57:     // Apply to avatar
  58:     const aAction = this.avatarActions[clip];
  59:     if (aAction) {
  60:       if (this.currentClip !== clip) {
  61:         this.avatarMixer.stopAllAction();
  62:         aAction.reset().setEffectiveWeight(1).setLoop(THREE.LoopOnce, 1).play();
  63:         aAction.clampWhenFinished = true;
  64:         aAction.paused = true;
  65:       }
  66:       aAction.time = t;
  67:       this.avatarMixer.update(0);
  68:     }
  69: 
  70:     // Apply to chair
  71:     const cAction = this.chairActions[clip];
  72:     if (cAction) {
  73:       if (this.currentClip !== clip) {
  74:         this.chairMixer.stopAllAction();
  75:         cAction.reset().setEffectiveWeight(1).setLoop(THREE.LoopOnce, 1).play();
  76:         cAction.clampWhenFinished = true;
  77:         cAction.paused = true;
  78:       }
  79:       cAction.time = t;
  80:       this.chairMixer.update(0);
  81:     }
  82: 
  83:     this.currentClip = clip;
  84:     this.currentTime = t;
  85:   }
  86: 
  87:   public playGreeting(): number {
  88:     if (this.mode === "sequence") return this.revision;
  89:     this.revision++;
  90:     this.queue = [
  91:       { clip: "notice_visitor", from: 0, to: 0.6, elapsed: 0 },
  92:       { clip: "turn_to_visitor", from: 0, to: 1.2, elapsed: 0 },
  93:       { clip: "greeting_nod", from: 0, to: 0.9, elapsed: 0 },
  94:       { clip: "return_to_work", from: 0, to: 1.3, elapsed: 0 },
  95:     ];
  96:     this.mode = "sequence";
  97:     if (this.queue.length > 0) {
  98:       const first = this.queue[0];
  99:       if (first) this.applyClip(first.clip, first.from);
 100:     }
 101:     return this.revision;
 102:   }
 103: 
 104:   public cancel(): number {
 105:     if (this.mode === "safe-return" || this.mode === "coding") return this.revision;
 106:     this.revision++;
 107:     const { currentClip, currentTime } = this;
 108:     const segments: QueueSegment[] = [];
 109: 
 110:     if (currentClip === "notice_visitor") {
 111:       segments.push({ clip: "notice_visitor", from: currentTime, to: 0, elapsed: 0 });
 112:     } else if (currentClip === "turn_to_visitor") {
 113:       segments.push(
 114:         { clip: "turn_to_visitor", from: currentTime, to: 0, elapsed: 0 },
 115:         { clip: "notice_visitor", from: 0.6, to: 0, elapsed: 0 }
 116:       );
 117:     } else if (currentClip === "greeting_nod") {
 118:       segments.push(
 119:         { clip: "greeting_nod", from: currentTime, to: 0, elapsed: 0 },
 120:         { clip: "turn_to_visitor", from: 1.2, to: 0, elapsed: 0 },
 121:         { clip: "notice_visitor", from: 0.6, to: 0, elapsed: 0 }
 122:       );
 123:     } else if (currentClip === "return_to_work") {
 124:       // Return is already moving toward rest pose, so complete the return
 125:       segments.push({ clip: "return_to_work", from: currentTime, to: 1.3, elapsed: 0 });
 126:     }
 127: 
 128:     this.queue = segments;
 129:     this.mode = segments.length ? "safe-return" : "coding";
 130:     if (this.queue.length > 0) {
 131:       const first = this.queue[0];
 132:       if (first) this.applyClip(first.clip, first.from);
 133:     } else {
 134:       this.applyClip("coding_idle", 0);
 135:     }
 136:     return this.revision;
 137:   }
 138: 
 139:   /**
 140:    * Instant settlement on Skip, Escape, or immediate route navigation.
 141:    * Settles within 1 frame (≤50ms), bypassing reverse animations.
 142:    */
 143:   public settle(): number {
 144:     this.revision++;
 145:     this.queue = [];
 146:     this.mode = "coding";
 147:     this.applyClip("coding_idle", 0);
 148:     return this.revision;
 149:   }
 150: 
 151:   public advance(dt: number) {
 152:     if (!Number.isFinite(dt) || dt < 0) return;
 153:     let remaining = dt;
 154: 
 155:     while (this.queue.length > 0) {
 156:       const seg = this.queue[0];
 157:       if (!seg) break;
 158:       const duration = Math.abs(seg.to - seg.from);
 159:       const take = Math.min(remaining, Math.max(0, duration - seg.elapsed));
 160:       seg.elapsed += take;
 161:       remaining -= take;
 162: 
 163:       const progress = duration > 0 ? seg.elapsed / duration : 1;
 164:       const sampleTime = seg.from + (seg.to - seg.from) * progress;
 165:       this.applyClip(seg.clip, sampleTime);
 166: 
 167:       if (seg.elapsed < duration - 1e-6) {
 168:         return;
 169:       }
 170: 
 171:       this.applyClip(seg.clip, seg.to);
 172:       this.queue.shift();
 173: 
 174:       if (this.queue.length > 0) {
 175:         const next = this.queue[0];
 176:         if (next) this.applyClip(next.clip, next.from);
 177:       } else {
 178:         this.mode = "coding";
 179:         this.applyClip("coding_idle", 0);
 180:       }
 181: 
 182:       if (remaining < 1e-6) return;
 183:     }
 184: 
 185:     if (this.mode === "coding") {
 186:       const codingDuration = CLIP_DURATIONS["coding_idle"] || 6.0;
 187:       this.applyClip("coding_idle", (this.currentTime + remaining) % codingDuration);
 188:     }
 189:   }
 190: 
 191:   public getChairYawDeg(): number {
 192:     const q = this.chairRoot.getWorldQuaternion(new THREE.Quaternion());
 193:     q.multiply(this.restChair.clone().invert());
 194:     const forward = new THREE.Vector3(0, 0, -1).applyQuaternion(q);
 195:     return THREE.MathUtils.radToDeg(Math.atan2(-forward.x, -forward.z));
 196:   }
 197: 
 198:   public getBodyYawDeg(): number {
 199:     const q = this.bodyTurn.getWorldQuaternion(new THREE.Quaternion());
 200:     q.multiply(this.restBody.clone().invert());
 201:     const forward = new THREE.Vector3(0, 0, -1).applyQuaternion(q);
 202:     return THREE.MathUtils.radToDeg(Math.atan2(-forward.x, -forward.z));
 203:   }
 204: 
 205:   public getDiagnostics() {
 206:     return {
 207:       currentClip: this.currentClip,
 208:       currentTime: this.currentTime,
 209:       mode: this.mode,
 210:       queueLength: this.queue.length,
 211:       chairYawDeg: this.getChairYawDeg(),
 212:       bodyYawDeg: this.getBodyYawDeg(),
 213:     };
 214:   }
 215: }
