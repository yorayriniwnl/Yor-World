45: /**
46:  * Validates request origin to protect mutating admin endpoints from CSRF and cross-site tampering.
47:  */
48: export function validateRequestOrigin(request: Request): { valid: boolean; reason?: string } {
49:   const method = request.method.toUpperCase();
50:   // Safe read methods
51:   if (method === "GET" || method === "HEAD" || method === "OPTIONS") {
52:     return { valid: true };
53:   }
54: 
55:   // Check Sec-Fetch-Site if provided
56:   const secFetchSite = request.headers.get("sec-fetch-site");
57:   if (secFetchSite && secFetchSite === "cross-site") {
58:     return { valid: false, reason: "Cross-site request blocked by Sec-Fetch-Site" };
59:   }
60: 
61:   // Check Origin header
62:   const origin = request.headers.get("origin");
63:   if (!origin) {
64:     // If origin is missing, check referer as fallback
65:     const referer = request.headers.get("referer");
66:     if (!referer) {
67:       // In non-browser API environments (like curl or unit tests), allow if host matches
68:       return { valid: true };
69:     }
70:     try {
71:       const refererUrl = new URL(referer);
72:       const host = request.headers.get("host") || request.headers.get("x-forwarded-host");
73:       if (host && refererUrl.host !== host) {
74:         return { valid: false, reason: "Referer host mismatch" };
75:       }
76:       return { valid: true };
77:     } catch {
78:       return { valid: false, reason: "Invalid referer URL" };
79:     }
80:   }
81: 
82:   try {
83:     const originUrl = new URL(origin);
84:     const host = request.headers.get("host") || request.headers.get("x-forwarded-host");
85:     if (host && originUrl.host !== host) {
86:       return { valid: false, reason: `Origin host '${originUrl.host}' does not match server host '${host}'` };
87:     }
88:     return { valid: true };
89:   } catch {
90:     return { valid: false, reason: "Malformed origin header" };
91:   }
92: }

94: /**
95:  * Extracts session bearer token or cookie from request headers.
96:  */
97: export function extractAuthToken(request: Request): string | null {
98:   const authHeader = request.headers.get("authorization");
99:   if (authHeader) {
100:     const match = authHeader.match(/^Bearer\s+(.+)$/i);
101:     if (match && match[1]) {
102:       return match[1].trim();
103:     }
104:   }
105: 
106:   const cookieHeader = request.headers.get("cookie");
107:   if (cookieHeader) {
108:     // Look for standard Supabase token cookies: sb-access-token or sb-<project>-auth-token
109:     const cookies = cookieHeader.split(";").map((c) => c.trim());
110:     for (const cookie of cookies) {
111:       const [name, ...valParts] = cookie.split("=");
112:       const val = valParts.join("=");
113:       if (!name || !val) continue;
114: 
115:       if (name === "sb-access-token" || name === "yor-admin-token") {
116:         return decodeURIComponent(val);
117:       }
118:       if (name.startsWith("sb-") && name.endsWith("-auth-token")) {
119:         try {
120:           const parsed = JSON.parse(decodeURIComponent(val));
121:           if (Array.isArray(parsed) && parsed[0]) {
122:             return parsed[0];
123:           }
124:           if (parsed && typeof parsed.access_token === "string") {
125:             return parsed.access_token;
126:           }
127:         } catch { /* Standard SSR base64/chunked cookies are handled by createServerClient. */ }
128:       }
129:     }
130:   }
131: 
132:   return null;
133: }

135: /**
136:  * Authoritatively verifies owner identity, MFA assurance, and active role.
137:  * Never leaks internal error or table details to caller.
138:  */
139: export async function verifyOwner(request: Request): Promise<AuthResult> {
140:   // 1. Request-origin protection for mutations
141:   const originCheck = validateRequestOrigin(request);
142:   if (!originCheck.valid) {
143:     return {
144:       ok: false,
145:       status: 403,
146:       code: "FORBIDDEN_ORIGIN",
147:       message: "Forbidden: Cross-origin request rejected",
148:     };
149:   }
150: 
151:   // 2. Token extraction
152:   const token = extractAuthToken(request);
153:   if (!token && !request.headers.get("cookie")?.includes("sb-")) {
154:     return {
155:       ok: false,
156:       status: 401,
157:       code: "UNAUTHENTICATED",
158:       message: "Authentication required",
159:     };
160:   }
161: 
162:   // 3. Resolve user identity and MFA assurance
163:   let userId: string;
164:   let userEmail = "";
165:   let assurance: "aal1" | "aal2" = "aal1";
166: 
167:   // Check test registry first if configured
168:   const fixtureToken = isE2EFixture() && ["127.0.0.1", "localhost", "[::1]"].includes(new URL(request.url).hostname) && token === "rc3-fixture-owner-aal2";
169:   if (fixtureToken) {
170:     userId = "11111111-1111-1111-1111-111111111111";
171:     userEmail = "owner@yorworld.test";
172:     assurance = "aal2";
173:   } else if (token && isTestRuntime() && testUserRegistry?.has(token)) {
174:     const record = testUserRegistry.get(token)!;
175:     userId = record.user.id;
176:     userEmail = record.user.email;
177:     assurance = record.user.aal;
178:   } else {
179:     // Check Supabase Auth
180:     try {
181:       const requestCookies = (request.headers.get("cookie") || "").split(";").flatMap((part) => {
182:         const at = part.indexOf("=");
183:         return at < 0 ? [] : [{ name: part.slice(0, at).trim(), value: part.slice(at + 1).trim() }];
184:       });
185:       const publicClient = createPublicServerClient({ getAll: () => requestCookies });
186:       const { data: authData, error: authError } = await publicClient.auth.getUser(token ?? undefined);
187: 
188:       if (authError || !authData?.user) {
189:         return {
190:           ok: false,
191:           status: 401,
192:           code: "UNAUTHENTICATED",
193:           message: "Invalid or expired authentication session",
194:         };
195:       }
196: 
197:       const user = authData.user;
198:       userId = user.id;
199:       userEmail = user.email || "";
200: 
201:       // Read AAL from cryptographically verified session claims, never user metadata.
202:       const { data: claimsData, error: claimsError } = await publicClient.auth.getClaims(token ?? undefined);
203:       if (claimsError || claimsData?.claims.sub !== userId) {
204:         return { ok: false, status: 401, code: "UNAUTHENTICATED", message: "Invalid authentication session" };
205:       }
206:       assurance = claimsData.claims.aal === "aal2" ? "aal2" : "aal1";
207:     } catch {
208:       return {
209:         ok: false,
210:         status: 401,
211:         code: "UNAUTHENTICATED",
212:         message: "Authentication verification failed",
213:       };
214:     }
215:   }
216: 
217:   // 4. Enforce MFA / AAL2 requirement
218:   // Per spec: MFA is not decorative. AAL2 is strictly required for owner administration.
219:   if (assurance !== "aal2") {
220:     return {
221:       ok: false,
222:       status: 403,
223:       code: "FORBIDDEN_MFA_REQUIRED",
224:       message: "Multi-factor authentication (AAL2) required",
225:     };
226:   }
227: 
228:   // 5. Query authoritative owner record from admin_users table
229:   // Never trust client role claims or email string alone.
230:   let adminRecord: AdminUserRecord | null = null;
231: 
232:   if (fixtureToken) {
233:     const rows=await (await getPlatformDb()).query("SELECT id,role,active FROM public.admin_users WHERE id=$1",[userId]);
234:     adminRecord=(rows.rows[0] as unknown as AdminUserRecord) ?? null;
235:   } else if (token && isTestRuntime() && testUserRegistry?.has(token)) {
236:     const record = testUserRegistry.get(token)!;
237:     adminRecord = record.adminRecord || null;
238:   } else {
239:     try {
240:       const adminClient = createAdminServiceRoleClient();
241:       const { data, error } = await adminClient
242:         .from("admin_users")
243:         .select("id, role, active")
244:         .eq("id", userId)
245:         .single();
246: 
247:       if (!error && data) {
248:         adminRecord = data as AdminUserRecord;
249:       }
250:     } catch {
251:       adminRecord = null;
252:     }
253:   }
254: 
255:   // 6. Check owner role authorization
256:   if (!adminRecord || adminRecord.role !== "owner") {
257:     return {
258:       ok: false,
259:       status: 403,
260:       code: "FORBIDDEN_NOT_OWNER",
261:       message: "Owner authorization required",
262:     };
263:   }
264: 
265:   // 7. Check immediate revocation
266:   // A revoked owner with a still-valid JWT must be denied immediately!
267:   if (!adminRecord.active) {
268:     return {
269:       ok: false,
270:       status: 403,
271:       code: "FORBIDDEN_REVOKED",
272:       message: "Owner authorization has been revoked",
273:     };
274:   }
275: 
276:   // Authorized active owner with AAL2
277:   return {
278:     ok: true,
279:     context: {
280:       userId,
281:       email: userEmail,
282:       assurance: "aal2",
283:       role: "owner",
284:       active: true,
285:     },
286:   };
287: }

289: /**
290:  * Enforces owner authorization on a request.
291:  * Throws OwnerAuthError on denial with typed status (401 or 403).
292:  */
293: export async function requireOwner(request: Request): Promise<OwnerContext> {
294:   const result = await verifyOwner(request);
295:   if (!result.ok) {
296:     throw new OwnerAuthError(result.status, result.code, result.message);
297:   }
298:   return result.context;
299: }

301: /**
302:  * Creates a sanitized JSON response for an authentication/authorization failure.
303:  * Ensures headers prevent private response caching.
304:  */
305: export function createAuthErrorResponse(failure: {
306:   status: 401 | 403;
307:   code: AuthFailureCode;
308:   message: string;
309: }): Response {
310:   return new Response(
311:     JSON.stringify({
312:       error: failure.message,
313:       code: failure.code,
314:     }),
315:     {
316:       status: failure.status,
317:       headers: {
318:         "Content-Type": "application/json",
319:         "Cache-Control": "no-store, no-cache, must-revalidate, proxy-revalidate",
320:         Pragma: "no-cache",
321:         Expires: "0",
322:         Vary: "Authorization, Cookie",
323:       },
324:     }
325:   );
