1: import { verifyOwner, createAuthErrorResponse } from "@/server/auth/require-owner";
2: import {
3:   publishRevision,
4:   readPublicPublication,
5:   readPublicationHistory,
6: } from "@/server/content/publish";
7: import { RevisionConflictError, ContentValidationError } from "@/server/content/revisions";
8: import { MediaValidationError } from "@/server/media/validate-upload";
9: 
10: export const dynamic = "force-dynamic";
11: 
12: export async function GET(request: Request) {
13:   const result = await verifyOwner(request);
14:   if (!result.ok) {
15:     return createAuthErrorResponse(result);
16:   }
17: 
18:   try {
19:   const active = await readPublicPublication();
20:   if (!active) return Response.json({ error: "Publication storage is unavailable." },{ status: 503 });
21:   const history = await readPublicationHistory();
22: 
23:   return Response.json(
24:     {
25:       currentRevision: active.revision,
26:       activePublication: active,
27:       history,
28:       authorizedActor: result.context.userId,
29:     },
30:     {
31:       headers: {
32:         "Cache-Control": "no-store",
33:         Vary: "Authorization, Cookie",
34:       },
35:     }
36:   );
37:   } catch {
38:     return Response.json({ error: "Private storage is temporarily unavailable." },{ status: 503,headers: { "Cache-Control": "no-store, private",Vary: "Authorization, Cookie" } });
39:   }
40: 
41: }
42: 
43: export async function POST(request: Request) {
44:   const result = await verifyOwner(request);
45:   if (!result.ok) {
46:     return createAuthErrorResponse(result);
47:   }
48: 
49:   try {
50:     const body = await request.json();
51: 
52:     if (typeof body.expectedRevision !== "number") {
53:       return Response.json(
54:         { error: "expectedRevision (number) is required" },
55:         { status: 422 }
56:       );
57:     }
58: 
59:     const publication = await publishRevision(
60:       {
61:         expectedRevision: body.expectedRevision,
62:         projectId: body.projectId,
63:       },
64:       result.context
65:     );
66: 
67:     return Response.json(
68:       {
69:         success: true,
70:         action: "publication_published",
71:         publication,
72:         actor: result.context.userId,
73:       },
74:       { status: 200 }
75:     );
76:   } catch (err) {
77:     if (err instanceof RevisionConflictError) {
78:       return Response.json(
79:         { error: err.message, code: err.code },
80:         { status: 409 }
81:       );
82:     }
83:     if (err instanceof ContentValidationError || err instanceof MediaValidationError) {
84:       return Response.json(
85:         { error: err.message, code: "UNPROCESSABLE_ENTITY" },
86:         { status: 422 }
87:       );
88:     }
89:     return Response.json(
90:       { error: "Private storage is temporarily unavailable." },
91:       { status: 503 }
92:     );
93:   }
94: }
