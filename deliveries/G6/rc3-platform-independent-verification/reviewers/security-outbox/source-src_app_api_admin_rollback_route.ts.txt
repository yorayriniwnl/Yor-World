1: import { verifyOwner, createAuthErrorResponse } from "@/server/auth/require-owner";
2: import { rollbackPublication } from "@/server/content/publish";
3: import { ContentValidationError } from "@/server/content/revisions";
4: import { MediaValidationError } from "@/server/media/validate-upload";
5: 
6: export const dynamic = "force-dynamic";
7: 
8: export async function POST(request: Request) {
9:   const result = await verifyOwner(request);
10:   if (!result.ok) {
11:     return createAuthErrorResponse(result);
12:   }
13: 
14:   try {
15:     const body = await request.json();
16: 
17:     if (typeof body.targetRevision !== "number") {
18:       return Response.json(
19:         { error: "targetRevision (number) is required" },
20:         { status: 422 }
21:       );
22:     }
23: 
24:     const publication = await rollbackPublication(body.targetRevision, result.context);
25: 
26:     return Response.json(
27:       {
28:         success: true,
29:         action: "publication_rollback",
30:         publication,
31:         actor: result.context.userId,
32:       },
33:       { status: 200 }
34:     );
35:   } catch (err) {
36:     if (err instanceof ContentValidationError || err instanceof MediaValidationError) {
37:       return Response.json(
38:         { error: err.message, code: "UNPROCESSABLE_ENTITY" },
39:         { status: 422 }
40:       );
41:     }
42:     return Response.json(
43:       { error: "Private storage is temporarily unavailable." },
44:       { status: 503 }
45:     );
46:   }
47: }
