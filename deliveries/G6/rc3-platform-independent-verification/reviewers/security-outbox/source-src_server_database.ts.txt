36: export async function getPlatformDb(): Promise<QueryableDb> {
37:   if (overrideDb) return overrideDb;
38:   if (isE2EFixture()) {
39:     const fixturePath = process.env.YOR_TEST_DATABASE_PATH!;
40:     const databases = fixtureState.__yorRc3FixtureDatabases ??= new Map();
41:     let fixtureDb = databases.get(fixturePath);
42:     if (!fixtureDb) {
43:       fixtureDb = import("./test-fixture").then(({ createFixtureDb }) => createFixtureDb());
44:       databases.set(fixturePath, fixtureDb);
45:     }
46:     return fixtureDb;
47:   }
48:   const connectionString = process.env.DATABASE_URL;
49:   if (!connectionString) throw new Error("Durable database is not configured.");
50:   pool ??= new Pool({ connectionString, max: 5, connectionTimeoutMillis: 5000, idleTimeoutMillis: 10000 });
51:   const currentPool = pool;
52:   return {
53:     query: async (sql, params = []) => ({ rows: (await currentPool.query(sql, params)).rows }),
54:     transaction: async <T>(run: (tx: QueryableDb) => Promise<T>): Promise<T> => {
55:       const client = await currentPool.connect();
56:       try {
57:         await client.query("BEGIN");
58:         const result = await run(clientHandle(client));
59:         await client.query("COMMIT");
60:         return result;
61:       } catch (error) {
62:         try { await client.query("ROLLBACK"); } catch { /* Preserve original failure. */ }
63:         throw error;
64:       } finally {

66:       }
67:     },
68:   };
