4: -- 1. Setup Auth Schema, Roles, and Context Functions if not present (PostgreSQL / Supabase compatibility)
5: CREATE SCHEMA IF NOT EXISTS auth;
6: 
7: DO $$
8: BEGIN
9:   IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'anon') THEN
10:     CREATE ROLE anon NOLOGIN;
11:   END IF;
12:   IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'authenticated') THEN
13:     CREATE ROLE authenticated NOLOGIN;
14:   END IF;
15:   IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'service_role') THEN
16:     CREATE ROLE service_role NOLOGIN;
17:   END IF;
18: END $$;

20: -- Simulated / fallback auth functions for test environments if running outside live Supabase
21: CREATE TABLE IF NOT EXISTS auth.users (
22:   id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
23:   email text UNIQUE,
24:   raw_app_meta_data jsonb DEFAULT '{}'::jsonb,
25:   raw_user_meta_data jsonb DEFAULT '{}'::jsonb,
26:   created_at timestamptz DEFAULT now()
27: );
28: 
29: CREATE OR REPLACE FUNCTION auth.uid()
30: RETURNS uuid
31: LANGUAGE sql
32: STABLE
33: AS $$
34:   SELECT NULLIF(current_setting('request.jwt.claim.sub', true), '')::uuid;
35: $$;
36: 
37: CREATE OR REPLACE FUNCTION auth.jwt()
38: RETURNS jsonb
39: LANGUAGE sql
40: STABLE
41: AS $$
42:   SELECT COALESCE(NULLIF(current_setting('request.jwt.claims', true), '')::jsonb, '{}'::jsonb);
43: $$;

54: -- 3. Security Definer Helper Functions (Section 9)
55: CREATE OR REPLACE FUNCTION public.is_active_owner()
56: RETURNS boolean
57: LANGUAGE sql
58: STABLE
59: SECURITY DEFINER
60: SET search_path = public
61: AS $$
62:   SELECT EXISTS (
63:     SELECT 1 FROM public.admin_users
64:     WHERE admin_users.id = auth.uid()
65:       AND admin_users.role = 'owner'
66:       AND admin_users.active = true
67:   );
68: $$;
69: 
70: CREATE OR REPLACE FUNCTION public.is_active_owner_with_aal2()
71: RETURNS boolean
72: LANGUAGE sql
73: STABLE
74: SECURITY DEFINER
75: SET search_path = public
76: AS $$
77:   SELECT EXISTS (
78:     SELECT 1 FROM public.admin_users
79:     WHERE admin_users.id = auth.uid()
80:       AND admin_users.role = 'owner'
81:       AND admin_users.active = true
82:   ) AND COALESCE((auth.jwt() ->> 'aal'), '') = 'aal2';
83: $$;

237: -- 6. Least-Privilege Grants Model (Section 8)
238: -- Revoke default public schema permissions
239: REVOKE ALL ON SCHEMA public FROM PUBLIC;
240: REVOKE ALL ON ALL TABLES IN SCHEMA public FROM PUBLIC;
241: 
242: GRANT USAGE ON SCHEMA public TO anon, authenticated, service_role;
243: GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO authenticated, service_role;
244: 
245: -- Public table: published_content
246: GRANT SELECT ON public.published_content TO anon, authenticated;
247: GRANT ALL ON public.published_content TO service_role;
248: 
249: -- Authenticated table grants (controlled strictly by RLS)
250: GRANT SELECT ON public.admin_users TO authenticated;
251: GRANT SELECT, INSERT, UPDATE, DELETE ON public.projects TO authenticated;
252: GRANT SELECT, INSERT, UPDATE, DELETE ON public.project_revisions TO authenticated;
253: GRANT SELECT, INSERT, UPDATE, DELETE ON public.evidence_records TO authenticated;
254: GRANT SELECT, INSERT, UPDATE, DELETE ON public.media_assets TO authenticated;
255: GRANT SELECT, INSERT, UPDATE, DELETE ON public.site_revisions TO authenticated;

257: GRANT SELECT, INSERT, UPDATE, DELETE ON public.published_content TO authenticated;

259: GRANT SELECT, INSERT ON public.audit_events TO authenticated;
